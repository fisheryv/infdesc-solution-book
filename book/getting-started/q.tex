\begin{chapex}
The video-sharing website \textit{YouTube} assigns to each video a unique identifier, which is a string of 11 characters from the set
\[ \{ \mathtt{A}, \mathtt{B}, \dots, \mathtt{Z}, \mathtt{a}, \mathtt{b}, \dots, \mathtt{z}, \mathtt{0}, \mathtt{1}, \mathtt{2}, \mathtt{3}, \mathtt{4}, \mathtt{5}, \mathtt{6}, \mathtt{7}, \mathtt{8}, \mathtt{9}, \text{\texttt{-}}, \text{\texttt{\_}} \} \]
This string is actually a natural number expressed in base-64, where the characters in the above set represent the numbers $0$ through $63$, in the same order---thus $\mathtt{C}$ represents $2$, $\mathtt{c}$ represents $28$, $\mathtt{3}$ represents $55$, and \texttt{\_} represents $63$. According to this schema, find the natural number whose base-64 expansion is $\mathtt{dQw4w9WgXcQ}$, and find the base-64 expansion of the natural number $7\,159\,047\,702\,620\,056\,984$.
\end{chapex}

\begin{mysolution}
    \textbf{base-64 to base-10:}
    \begin{align*}
        \mathtt{dQw4w9WgXcQ} =& \mathtt{d}\cdot 36^{10}+\mathtt{Q}\cdot 36^{9}+\mathtt{w}\cdot 36^{8}+\mathtt{4}\cdot 36^{7}+\mathtt{w}\cdot 36^{6}+\mathtt{9}\cdot 36^{5}+\\
        &\mathtt{W}\cdot 36^{4}+\mathtt{g}\cdot 36^{3}+\mathtt{X}\cdot 36^{2}+\mathtt{c}\cdot 36^{1}+\mathtt{Q}\cdot 36^{0}\\
        = & 29 \cdot 36^{10}+16 \cdot 36^{9}+48 \cdot 36^{8}+56 \cdot 36^{7}+48 \cdot 36^{6}+61 \cdot 36^{5}+\\
        &22 \cdot 36^{4}+32 \cdot 36^{3}+23 \cdot 36^{2}+28 \cdot 36^{1}+16 \cdot 36^{0}\\
        = & 33\,736\,714\,463\,647\,725\,328
    \end{align*}
    So, the natural number whose base-64 expansion is $\mathtt{dQw4w9WgXcQ}$ is $33\,736\,714\,463\,647\,725\,328$.
    
    \textbf{base-10 to base-64:}

\begin{chapex}
\label{cqProductInZSqrtTwoEqualsInteger}
Let $a, b, c, d \in \mathbb{Z}$. Under what conditions is $(a+b\sqrt{2})(c+d\sqrt{2})$ an integer?
\end{chapex}

\begin{mysolution}
    We know that
    \[(a+b\sqrt{2})(c+d\sqrt{2}) = ac + 2bd + (ad+bc)\sqrt{2}\]
    Since $a, b, c, d \in \mathbb{Z}$, $ac+2bd$ and $ad+bc$ are integers. But $\sqrt{2}$ is irrational, to make $ac + 2bd + (ad+bc)\sqrt{2}$ an integer, $(ad+bc)$ must be $0$. 
    
    Therefore, when $(ad+bc) = 0$, $(a+b\sqrt{2})(c+d\sqrt{2})$ is an integer.
\end{mysolution}

\begin{chapex}
\label{cqSumOfRemaindersModulo3}
Suppose an integer $m$ leaves a remainder of $i$ when divided by $3$, and an integer $n$ leaves a remainder of $j$ when divided by $3$, where $0 \le i,j < 3$. Prove that $m+n$ and $i+j$ leave the same remainder when divided by $3$.
\end{chapex}

\begin{myproof}
    According to the problem statement, we have
    \[m = 3k + i \quad \text{ and } \quad n = 3l + j\]
    where $k, l \in \mathbb{Z}$ and $0 \le i,j < 3$.
    \[m+n = (3k + i)+(3l + j) = 3(k+l) + (i+j)\]
    Since $3(k+l)$ is multiple of $3$, so $3(k+l) + (i+j) \equiv (i+j) \pmod 3$. 
    
    Therefore, $m+n$ and $i+j$ leave the same remainder when divided by $3$.
\end{myproof}

\begin{chapex}
\label{cqRemaindersOfNSquared}
What are the possible remainders of $n^2$ when divided by $3$, where $n \in \mathbb{Z}$?
\end{chapex}

\begin{myproof}
    Let $n \in \mathbb{Z}$. By the division theorem, one of the following must be true for some $k \in \mathbb{Z}$:
    \[
    n=3k \quad \text{or} \quad n=3k+1 \quad \text{or} \quad n=3k+2
    \]
    \begin{itemize}
    \item Suppose $n=3k$. Then
    \[
    n^2 = (3k)^2 = 9k^2 = 3 \cdot (3k^2)
    \]
    So $n^2$ leaves a remainder of $0$ when divided by $3$.
    \item Suppose $n=3k+1$. Then
    \[
    n^2 = (3k+1)^2 = 9k^2+6k+1 = 3(3k^2+2k)+1
    \]
    So $n^2$ leaves a remainder of $1$ when divided by $3$.
    \item Suppose $n=3k+2$. Then
    \[
    n^2 = (3k+2)^2 = 9k^2+12k+4 = 3(3k^2+4k+1)+1
    \]
    So $n^2$ leaves a remainder of $1$ when divided by $3$.
    \end{itemize}
    In all cases, $n^2$ leaves a remainder of $0$ or $1$ when divided by $3$.
\end{myproof}

\end{definition}

\Crefrange{cqAlgebraicNumbersBegin}{cqAlgebraicNumbersEnd} concern \textit{algebraic numbers}, defined next.

\begin{definition}[Algebraic number]
\label{defAlgebraicNumber}
\index{number!algebraic}
\index{algebraic number}
A complex number $\alpha$ is \textbf{algebraic} if $p(\alpha) = 0$ for some nonzero polynomial $p(x)$ over $\mathbb{Q}$.
\end{definition}

\begin{mysolution}
    \Crefrange{cqAlgebraicNumbersBegin}{cqAlgebraicNumbersEnd} are existence prooves, which only requires constructing a nonzero polynomial that meets the requirements.
\end{mysolution}

\begin{chapex}
\label{cqAlgebraicNumbersBegin}
Let $x$ be a rational number. Prove that $x$ is an algebraic number.
\end{chapex}

\begin{myproof}
    Since $x$ is a rational number, so it can be written as $x = \frac{u}{v}$ for some integers $u$ and $v$ with $v \ne 0$. Let nonzero polynomial $p(x)$ over $\mathbb{Q}$ be
    \[
    p(x) = vx - u
    \]
    Then $p(\frac{u}{v}) = v \cdot \frac{u}{v}-u = 0$. 
    
    We have shown that there exists a non-zero polynomial $p(x)$ over $\mathbb{Q}$ such that $p(\alpha) = 0$ for any $\alpha \in \mathbb{Q}$. Therefore, $x \in \mathbb{Q}$ is an algebraic number.
\end{myproof}

\begin{chapex}
Prove that $\sqrt{2}$ is an algebraic number.
\end{chapex}

\begin{myproof}
    Let nonzero polynomial $p(x)$ over $\mathbb{Q}$ be
    \[
    p(x) = x^2 - 2
    \]
    Then $p(\sqrt{2}) = (\sqrt{2})^2 - 2 = 0$. 
    
    We have shown that there exists a non-zero polynomial $p(x)$ over $\mathbb{Q}$ such that $p(\sqrt{2}) = 0$. Therefore, $\sqrt{2}$ is an algebraic number.
\end{myproof}

\begin{chapex}
Prove that $\sqrt{2} + \sqrt{3}$ is an algebraic number.
\end{chapex}

\begin{myproof}
    Let $x = \sqrt{2} + \sqrt{3}$, then
    \begin{align*}
        x^2 &= (\sqrt{2} + \sqrt{3})^2 = 5+2\sqrt{6}\\
        x^2-5 &= 2\sqrt{6} \\
        (x^2-5)^2 &= (2\sqrt{6})^2 = 24 \\
        x^4-10x^2-1 &=0
    \end{align*}
    So $\sqrt{2} + \sqrt{3}$ is one root of non-zero polynomial $p(x)=x^4-10x^2-1$.
    
    We have shown that there exists a non-zero polynomial $p(x)$ over $\mathbb{Q}$ such that $p(\sqrt{2} + \sqrt{3}) = 0$. Therefore, $\sqrt{2} + \sqrt{3}$ is an algebraic number.
\end{myproof}

\begin{chapex}
\label{cqAlgebraicNumbersEnd}
Prove that $x+yi$ is an algebraic number, where $x$ and $y$ are any two rational numbers.
\end{chapex}

\begin{myproof}
    Let $z = x+yi$, then
    \begin{align*}
        z^2 &= (x+yi)^2 = x^2-y^2+2xyi\\
        z^2-x^2+y^2 &= 2xyi \\
        (z^2-x^2+y^2)^2 &= (2xyi)^2 = -4x^2y^2 \\
        z^4-2(x^2-y^2)z^2+(x^2+y^2) &=0
    \end{align*}
    So $z = x+yi$ is one root of non-zero polynomial $p(z)=z^4-2(x^2-y^2)z^2+(x^2+y^2) =0$.
    
    We have shown that there exists a non-zero polynomial $p(x)$ over $\mathbb{Q}$ such that $p(z) = 0$ for any $x+yi$ where $x,y \in \mathbb{Q}$. Therefore, $z = x+yi$ is an algebraic number.
\end{myproof}

\Crefrange{cqModulusArgumentBegin}{cqModulusArgumentEnd} concern the \textit{modulus} and \textit{argument} of a complex number.

\begin{chapex}
\label{cqModulusArgumentBegin}
\index{modulus}
\index{argument}
\index{complex number!modulus and argument}
Let $z \in \Cplx$.
\begin{enumerate}[(a)]
\item Show that there exist $r,\theta \in \Real$ such that $z = r\cos(\theta) + r\sin(\theta) i$.
\item Show that if $r_1,r_2,\theta_1,\theta_2 \in \Real$ are such that $z=r_1\cos(\theta_1) + r_1\sin(\theta_1) i = r_2\cos(\theta_2) + r_2\sin(\theta_2) i$, then $|r_1| = |r_2|$, and if $z \ne 0$, then $\theta_2 = \theta_1 + n \cdot 2\pi$ for some $n \in \Int$.
\item Conclude that if $z \ne 0$ then there is exactly one $r \in [0,\infty)$ and exactly one $\theta \in (-\pi,\pi]$ such that $z=r\cos(\theta) + r\sin(\theta) i$.
\end{enumerate}
\end{chapex}

\begin{myproof}
    \begin{enumerate}[(a)]
    \item 
        Let $z=x+yi$, where $x,y \in \mathbb{R}$. There are two cases:
        \begin{itemize}  
            \item If $z=0$. Let $r=0$ and $\theta=0$, then
            \[z = 0\cos(0) + 0\sin(0) i = 0\]
            \item If $z \ne 0$. Let $r = \sqrt{x^2+y^2} > 0$, then
            \[\left(\dfrac{x}{r}\right)^2+\left(\dfrac{y}{r}\right)^2=1\]
            Hence the point $\left(\dfrac{x}{r},\dfrac{y}{r}\right)$ lies on the unit circle, so there exists $\theta \in \Real$ such that
            \[\cos \theta = \dfrac{x}{r} \enspace, \enspace \sin \theta = \dfrac{y}{r} \]
            Therefore, $z = x+yi = r \cos(\theta) + r \sin(\theta) i$.
        \end{itemize}
    In conclusion, there always exist $r,\theta \in \Real$ such that $z = r\cos(\theta) + r\sin(\theta) i$.
    \item Assume $r_1, r_2 >0$, since $z=r_1\cos(\theta_1) + r_1\sin(\theta_1) i = r_2\cos(\theta_2) + r_2\sin(\theta_2) i$, then
    \[|z|^2 = r_1^2\cos^2(\theta_1)+r_1^2\sin^2(\theta_1) = r_1^2\]
    \[|z|^2 = r_2^2\cos^2(\theta_2)+r_2^2\sin^2(\theta_2) = r_2^2\]
    So $r_1^2 = r_2^2$, this gives $|r_1| = |r_2|$.

    Since $|r_1| = |r_2|$, we have
    \[\cos(\theta_1) + \sin(\theta_1) i = \cos(\theta_2) + \sin(\theta_2) i\]
    then
    \[\cos(\theta_1) = \cos(\theta_2) \enspace \text{and} \enspace \sin(\theta_1) = \sin(\theta_2)\]
    therefore $\theta_1 = \theta_2 + n \cdot 2\pi$.
    \item Let $z \ne 0$
        \begin{itemize}
            \item \textbf{Existence: }Let $r=|z|>0$. By part (a), there exists $\theta_0 \in \Real$ such that
            \[z=r(\cos(\theta_0)+\sin(\theta_0) i)\]
            By part (b), there exists some $n \in \mathbb{Z}$ such that
            \[\theta = \theta_0+n \cdot 2\pi \in (-\pi, \pi]\]
            Then
            \[z = r\cos(\theta)+r\sin(\theta) i\]
            where $r \in [0,+\infty)$ and $\theta \in (-\pi, \pi]$.
            \item \textbf{Uniqueness: }Suppose
            \[z = r_1\cos(\theta_1)+r_1\sin(\theta_1) i = r_2\cos(\theta_2)+r_2\sin(\theta_2) i\]
            where $r_1, r_2 \in [0,+\infty)$ and $\theta_1, \theta_2 \in (-\pi, \pi]$. 
            
            By part (b), we have
            \[|z| = r_1 = r_2 > 0 \enspace \text{and} \enspace \theta_1 = \theta_2+n \cdot 2\pi\]
            Since $\theta_1, \theta_2 \in (-\pi, \pi]$, we have \[\theta_1-\theta_2 \in (-2\pi, 2\pi)\]
            so the only multiple of $2\pi$ in this interval is $0$. Hence $\theta_1 = \theta_2$.
        \end{itemize}
    \end{enumerate}
\end{myproof}

\begin{definition}[Modulus and argument of a complex number]
\label{defModulusArgument}
Let $z \in \Cplx$. The \textbf{modulus} $|z|$ and (\textbf{principal}) \textbf{argument} $\mathrm{arg}(z)$ of $z$ are the elements $|z| \in [0,\infty)$ and $\mathrm{arg}(z) \in (-\pi,\pi]$ such that
\[ z = |z| \cos(\mathrm{arg}(z)) + |z| \sin(\mathrm{arg}(z)) i \]
where we set $\mathrm{arg}(z)=0$ if $z=0$.

Note that the modulus and argument exist and are unique by \Cref{cqModulusArgumentBegin}.
\end{definition}

\begin{chapex}
\label{cqModulusArgumentOfProduct}
Let $z_1,z_2 \in \Cplx$, and for $k=1,2$ define $r_k = |z_k|$ and $\theta_k = \mathrm{arg}(z_k)$.

Prove that $z_1z_2 = r_1r_2\cos(\theta_1+\theta_2) + r_1r_2\sin(\theta_1+\theta_2)i$, and therefore express $|z_1z_2|$ and $\mathrm{arg}(z_1z_2)$ in terms of $r_1,r_2,\theta_1,\theta_2$.
\end{chapex}

\begin{myproof}
    Let $z_1 = r_1\cos(\theta_1) + r_1\sin(\theta_1) i$ and $z_2 = r_2\cos(\theta_2) + r_2\sin(\theta_2) i$. Then
    \begin{align*}
        z_1z_2 &= \big(r_1\cos(\theta_1) + r_1\sin(\theta_1) i\big)\big(r_2\cos(\theta_2) + r_2\sin(\theta_2) i\big)\\
        &= r_1r_2\cos(\theta_1)\cos(\theta_2) - r_1r_2\sin(\theta_1)\sin(\theta_2)+r_1r_2\cos(\theta_1)\sin(\theta_2) i+r_1r_2sin(\theta_1)\cos(\theta_2) i\\
        &= r_1r_2\big(\cos(\theta_1)\cos(\theta_2) - \sin(\theta_1)\sin(\theta_2)\big) + r_1r_2\big(\sin(\theta_1)\cos(\theta_2) + \cos(\theta_1)\sin(\theta_2)\big)i\\
        &= r_1r_2\cos(\theta_1+\theta_2) + r_1r_2\sin(\theta_1+\theta_2)i 
    \end{align*}
    So, $|z_1z_2| = r_1r_2$ and $\mathrm{arg}(z_1z_2) = \theta_1+\theta_2$.
\end{myproof}

\begin{chapex}
\label{cqModulusArgumentOfQuotient}
\label{cqModulusArgumentEnd}
Repeat \Cref{cqModulusArgumentOfProduct}, but for $\dfrac{z_1}{z_2}$ instead of $z_1z_2$.
\end{chapex}

\begin{myproof}
    Let $z_1 = r_1\cos(\theta_1) + r_1\sin(\theta_1) i$ and $z_2 = r_2\cos(\theta_2) + r_2\sin(\theta_2) i$. Then
    \begin{align*}
        \dfrac{z_1}{z_2} &= \dfrac{r_1\cos(\theta_1) + r_1\sin(\theta_1) i}{r_2\cos(\theta_2) + r_2\sin(\theta_2) i}\\
        &= \dfrac{\big(r_1\cos(\theta_1) + r_1\sin(\theta_1) i\big)\big(r_2\cos(\theta_2) - r_2\sin(\theta_2) i\big)}{\big(r_2\cos(\theta_2) + r_2\sin(\theta_2) i\big)\big(r_2\cos(\theta_2) - r_2\sin(\theta_2) i\big)}\\
        &= \dfrac{r_1r_2\cos(\theta_1)\cos(\theta_2) + r_1r_2\sin(\theta_1)\sin(\theta_2) + r_1r_2\sin(\theta_1)\cos(\theta_2) i - r_1r_2\cos(\theta_1)\sin(\theta_2) i }{\big(r_2\cos(\theta_2)\big)^2 - \big(r_2\sin(\theta_2)i\big)^2}\\
        &= \dfrac{r_1r_2\big(\cos(\theta_1)\cos(\theta_2) + \sin(\theta_1)\sin(\theta_2)\big) + r_1r_2\big(\sin(\theta_1)\cos(\theta_2)-\cos(\theta_1)\sin(\theta_2)\big) i}{r_2^2\cos^2(\theta_2) + r_2^2\sin^2(\theta_2)}\\
        &= \dfrac{r_1r_2\cos(\theta_1-\theta_2) + r_1r_2\sin(\theta_1-\theta_2) i}{r_2^2}\\
        &= \dfrac{r_1}{r_2}\cos(\theta_1-\theta_2) + \dfrac{r_1}{r_2}\sin(\theta_1-\theta_2) i
    \end{align*}
    So, $\left|\dfrac{z_1}{z_2}\right| = \dfrac{r_1}{r_2}$ and $\mathrm{arg}\left(\dfrac{z_1}{z_2}\right) = \theta_1-\theta_2$.
\end{myproof}

\begin{chapex}
\label{cqQuadraticFormulaOverC}
Prove a general version of the quadratic formula for quadratic polynomials over $\Cplx$.
\end{chapex}

\begin{myproof}
    Let $p(x) = ax^2+bx+c$, where $a,b,c \in \Cplx$ and $a \ne 0$. We prove that the roots of $p$ are given by
    $$ \dfrac{-b \pm \sqrt{\Delta}}{2a} $$
    where $\sqrt{\Delta} \in \Cplx$ satisfies $\Delta = b^2-4ac$.
    \begin{align*}
        p(x) &= ax^2+bx+c\\
             &= a\left(x^2+\dfrac{b}{a}x+\dfrac{c}{a}\right)\\
             &= a\left[\left(x+\dfrac{b}{2a}\right)^2-\dfrac{b^2-4ac}{4a^2}\right]
    \end{align*}
    Let $\Delta = b^2-4ac$, and choose $\sqrt{\Delta} \in \Cplx$, then
    \begin{align*}
        p(x) &= a\left[\left(x+\dfrac{b}{2a}\right)^2-\left(\dfrac{\sqrt{\Delta}}{2a}\right)^2\right] \\
        &=a\left(x+\dfrac{b}{2a}+\dfrac{\sqrt{\Delta}}{2a}\right)\left(x+\dfrac{b}{2a}-\dfrac{\sqrt{\Delta}}{2a}\right)
    \end{align*}
    Therefore, $p(x)=0$ if and only if
    $$x+\dfrac{b}{2a}+\dfrac{\sqrt{\Delta}}{2a}=0 \quad\text{or}\quad x+\dfrac{b}{2a}-\dfrac{\sqrt{\Delta}}{2a}=0$$
    Thus, the roots of $p(x)$ are
    $$x=\dfrac{-b \pm \sqrt{\Delta}}{2a}, \quad \sqrt{\Delta}=\sqrt{b^2-4ac} \in \Cplx$$
\end{myproof}

\subsection*{True--False questions}

\tfquestiontext{cqGettingStartedTFBegin}{cqGettingStartedTFEnd}

\begin{chapex} % False
\label{cqGettingStartedTFBegin}
Every integer is a natural number.
\end{chapex}

\begin{mysolution}
    \textbf{False}

    Because $\mathbb{N} \subset \mathbb{Z}$, not all integers are natural numbers. For example, $-1$ is an integer but is not a natural number.
\end{mysolution}

\begin{chapex} % True
Every integer is a rational number.
\end{chapex}

\begin{mysolution}
    \textbf{True}

    Because $\mathbb{Z} \subset \mathbb{Q}$, any integer $p$ can be written as $\frac{p}{1}$, so all integers are rational numbers.
\end{mysolution}

\begin{chapex} % True
Every integer divides zero.
\end{chapex}

\begin{mysolution}
    \textbf{True}

    Because $\forall n \in \mathbb{Z}, 0 = n \cdot 0$, according to the Definition 0.33, every integer divides zero.
\end{mysolution}

\begin{chapex} % True
Every integer divides its square.
\end{chapex}

\begin{mysolution}
    \textbf{True}

    Because $\forall n \in \mathbb{Z}, n^2 = n \cdot n$, According to the Definition 0.33, every integer divides its square.
\end{mysolution}

\begin{chapex} % True
The square of every rational number is a rational number.
\end{chapex}

\begin{mysolution}
    \textbf{True}

    Every rational number can be represented as a ratio of two integers $\frac{p}{q}, p,q \in \mathbb{Z}$. Its square is $\frac{p^2}{q^2}$, which is a rational number.
\end{mysolution}

\begin{chapex} % False
The square root of every positive rational number is a rational number.
\end{chapex}

\begin{mysolution}
    \textbf{False}

    We can give a counterexample: $2$ is a positive rational number, but its square root $\sqrt{2}$ is an irrational number.
\end{mysolution}

\begin{chapex} % False
When an integer $a$ is divided by a positive integer $b$, the remainder is always less than $a$.
\end{chapex}

\begin{mysolution}
    \textbf{False}

    According to the division theorem, $a = qb+r$, where $q, r \in \mathbb{Z}$ and the remainder $0 \le r < |b|$. If $a < 0$, then $r$ is greater than $a$. If $0 \le a < b$, then $r = a$.
\end{mysolution}

\begin{chapex} % True
\label{cqGettingStartedTFEnd}
Every quadratic polynomial over $\Real$ with no real roots has two distinct non-real roots.
\end{chapex}

\begin{mysolution}
    \textbf{True}

    By \Cref{cqQuadraticFormulaOverC}, if $\Delta > 0$, there are two distinct real roots; If $\Delta = 0$, there is one real root; If $\Delta < 0$, there are no real roots, but there are two distinct non-real roots.
\end{mysolution}

\subsection*{Always--Sometimes--Never questions}

\asnquestiontext{cqGettingStartedASNBegin}{cqGettingStartedASNEnd}

\begin{chapex} % Always
\label{cqGettingStartedASNBegin}
Let $n,b_1,b_2 \in \mathbb{N}$ with $1 < n < b_1 < b_2$. Then the base-$b_1$ expansion of $n$ is equal to the base-$b_2$ expansion of $n$.
\end{chapex}

\begin{mysolution}
    \textbf{Always}

    Given $1 < n < b_1 < b_2$, we have $n < b_1$ and $n < b_2$. In any base $b>n$, the integer $n$ is represented by a single digit equal to $n$ itself (since $0 \le n < b$). Therefore, the base-$b_1$ expansion of $n$ is the digit $n$, and the base-$b_2$ expansion of $n$ is also the digit $n$. Hence they are identical.
\end{mysolution}

\begin{chapex} % Never
Let $n,b_1,b_2 \in \mathbb{N}$ with $1 < b_1 < b_2 < n$. Then the base-$b_1$ expansion of $n$ is equal to the base-$b_2$ expansion of $n$.
\end{chapex}

\begin{mysolution}
    \textbf{Never}

    Given $1 < b_1 < b_2 < n$, we have $b_1 < n$ and $b_2 < n$. The integer $n$ can be represented as
    \[n = a_n \cdot b_1^n + a_{n-1} \cdot b_1^{n-1} + \cdots + a_1 \cdot b_1 + a_0\]
    \[n = a_n' \cdot b_2^n + a_{n-1}' \cdot b_2^{n-1} + \cdots + a_1' \cdot b_2 + a_0' \] 
    Because $b_1 \ne b_2$, the expansion of $n$ in base $b_1$ and $b_2$ are not the same.
\end{mysolution}

\begin{chapex} % Sometimes
Let $a,b,c \in \mathbb{Z}$ and suppose that $a$ divides $c$ and $b$ divides $c$. Then $ab$ divides $c$.
\end{chapex}

\begin{mysolution}
    \textbf{Sometimes}

    We can give a counterexample: $a=3, b=6, c=12$. Then $a$ divides $c$ and $b$ divides $c$, but $ab$ does not divide $c$.
\end{mysolution}

\begin{chapex} % Always
Let $a,b,c \in \mathbb{Z}$ and suppose that $a$ divides $c$ and $b$ divides $c$. Then $ab$ divides $c^2$.
\end{chapex}

\begin{mysolution}
    \textbf{Always}

    According to the problem statement, we have
    \[c = ma \quad \text{and} \quad c = nb \quad \text{where} \quad m,n \in \mathbb{Z}\]
    Then $c^2 = ma \cdot nb = (mn)\cdot(ab)$, which is a multiple of $ab$.

    Therefore, $ab$ divides $c^2$.
\end{mysolution}

\begin{chapex} % Always
Let $x,y \in \mathbb{Q}$ and let $a,b,c,d \in \mathbb{Z}$ with $cy+d \ne 0$. Then $\dfrac{ax+b}{cy+d} \in \mathbb{Q}$.
\end{chapex}

\begin{mysolution}
    \textbf{Always}

    $x,y \in \mathbb{Q}$, $x, y$ can be represented as $x=\dfrac{p}{q}$ and $y=\dfrac{r}{s}$, where $p,q,r,s \in \mathbb{Z}$. Then
    \[\dfrac{ax+b}{cy+d} = \dfrac{\dfrac{ap}{q}+b}{\dfrac{cr}{s}+d} = \dfrac{\dfrac{ap+bq}{q}}{\dfrac{cr+ds}{s}} = \dfrac{s(ap+bq)}{q(cr+ds)}\]
    Because $a, b,c,d,p,q,r,s \in \mathbb{Z}$, $\dfrac{ax+b}{cy+d}$ is a rational number.
\end{mysolution}

\begin{chapex} % Sometimes
Let $\dfrac{a}{b}$ be a rational number. Then $a \in \mathbb{Z}$ and $b \in \mathbb{Z}$.
\end{chapex}

\begin{mysolution}
    \textbf{Sometimes}

    We can give a counterexample: $\dfrac{\sqrt{2}}{\sqrt{2}} = 1$ is a rational number, but $\sqrt{2}$ is not a integer, even not a rational number.
\end{mysolution}

\begin{chapex} % Sometimes
Let $x \in \mathbb{R}$ and assume that $x^2 \in \mathbb{Q}$. Then $x \in \mathbb{Q}$.
\end{chapex}

\begin{mysolution}
    \textbf{Sometimes}

    We can give a counterexample: Let $x=\sqrt{2}$, $x^2=2$ is a rational number, but $\sqrt{2}$ is not a rational number.
\end{mysolution}

\begin{chapex} % Always
Let $x \in \mathbb{R}$ and assume that $x^2+1 \in \mathbb{Q}$ and $x^5+1 \in \mathbb{Q}$. Then $x \in \mathbb{Q}$.
\end{chapex}

\begin{mysolution}
    \textbf{Always}

    Given that $x^2 + 1 \in \mathbb{Q}$ and $x^5 + 1 \in \mathbb{Q}$, we can immediately deduce that $x^2 \in \mathbb{Q}$ and $x^5 \in \mathbb{Q}$.

    Let $x^2 = a$ and $x^5 = b$. 

    \begin{itemize}
        \item If $a = 0$, then $x = 0$. Since $0$ is a rational number, $x \in \mathbb{Q}$ is true.
        \item If $a \ne 0$, then $x = \dfrac{x^5}{x^4} = \dfrac{x^5}{(x^2)^2}=\dfrac{b}{a^2}$. Since $b \in \mathbb{Q}$, $a^2 \in \mathbb{Q}$, and $a^2 \neq 0$, $x \in \mathbb{Q}$ is true.
    \end{itemize}
    
    Therefore, $x \in \mathbb{Q}$ is always true.
\end{mysolution}

\begin{chapex} % Always
\label{cqGettingStartedASNEnd}
Let $p(x) = ax^2+bx+c$ be a polynomial with $a,b,c \in \mathbb{R}$ and $a \ne 0$, and suppose that $u+vi$ be a complex root of $p(x)$ with $v \ne 0$. Then $u-vi$ is a root of $p(x)$.
\end{chapex}

\begin{mysolution}
    \textbf{Always}

    By \Cref{cqQuadraticFormulaOverC}, $p(x)$ has two roots
    \[x=\dfrac{-b \pm \sqrt{\Delta}}{2a}, \quad \sqrt{\Delta}=\sqrt{b^2-4ac} \in \Cplx\]
    If $u+vi$ is one complex root of $p(x)$, we can deduce that $\Delta<0$, then $\sqrt{\Delta}$ can be written as $\sqrt{-\Delta} i$. Thus
    \[x=\dfrac{-b \pm \sqrt{\Delta}}{2a}=\dfrac{-b \pm \sqrt{-\Delta} i}{2a}, \quad \Delta < 0\]

    Let $u = \dfrac{-b}{2a}$ and $v=\dfrac{\sqrt{-\Delta}}{2a}$, the roots can be written as
    \[x = u \pm vi\]
    Therefore, $u-vi$ is a root of $p(x)$.
\end{mysolution}
